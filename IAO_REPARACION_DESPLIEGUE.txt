SHA256: b9eb519504659c94c2faf59c5579f26292eb82508239af1e09fd5a8b6c1a07cd
1: """Minimal supervisor: restarts the runtime on crash with backoff; HALTs after too many restarts.
2: On SIGTERM/SIGINT it stops the child too (no orphans) and exits 0."""
3: import signal
4: import subprocess
5: import os
6: import sys
7: import time
8: from types import FrameType
9: 
10: 
11: def supervise(cmd: list[str], max_restarts: int = 5, window: float = 60.0) -> int:
12:     restarts: list[float] = []
13:     child: dict[str, subprocess.Popen[bytes] | None] = {"p": None}
14:     stopping = {"v": False}
15: 
16:     def stop(_s: int, _f: FrameType | None) -> None:
17:         stopping["v"] = True
18:         p = child["p"]
19:         if p is not None and p.poll() is None:
20:             p.terminate()
21: 
22:     signal.signal(signal.SIGTERM, stop)
23:     signal.signal(signal.SIGINT, stop)
24:     while True:
25:         p = subprocess.Popen(cmd, env={**os.environ, "IAO_LLM_ROUTER": "1"})
26:         child["p"] = p
27:         if stopping["v"]:
28:             p.terminate()
29:         code = p.wait()
30:         if stopping["v"] or code == 0:
31:             return 0
32:         now = time.monotonic()
33:         restarts = [t for t in restarts if now - t < window] + [now]
34:         sys.stderr.write(f'{{"event":"runtime_exit","code":{code},"restarts_in_window":{len(restarts)}}}\n')
35:         if len(restarts) > max_restarts:
36:             sys.stderr.write('{"event":"HALT","reason":"restart_limit"}\n')
37:             return 1
38:         time.sleep(min(0.25 * 2 ** len(restarts), 5))
39: 
40: 
41: if __name__ == "__main__":
42:     raise SystemExit(supervise([sys.executable, "-m", "omega.server"]))
